    23→// ── grain/vignette shader ──
    24→const HorrorShader = {
    25→  uniforms: {
    26→    tDiffuse: { value: null as THREE.Texture | null },
    27→    uTime: { value: 0 },
    28→    uDamage: { value: 0 },
    29→    uThreat: { value: 0 },
    30→  },
    31→  vertexShader: /* glsl */ `
    32→    varying vec2 vUv;
    33→    void main() {
    34→      vUv = uv;
    35→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    36→    }
    37→  `,
    38→  fragmentShader: /* glsl */ `
    39→    uniform sampler2D tDiffuse;
    40→    uniform float uTime;
    41→    uniform float uDamage;
    42→    uniform float uThreat;
    43→    varying vec2 vUv;
    44→    float rand(vec2 co) { return fract(sin(dot(co, vec2(12.9898, 78.233))) * 43758.5453); }
    45→    void main() {
    46→      vec2 uv = vUv;
    47→      // تشويش خفيف
    48→      float grain = rand(uv * vec2(1024.0 + mod(uTime * 60.0, 100.0), 768.0)) * 0.055;
    49→      // انحراف لوني خفيف عند الحواف
    50→      vec2 dir = uv - 0.5;
    51→      float d2 = dot(dir, dir);
    52→      vec2 off = dir * d2 * 0.012;
    53→      float r = texture2D(tDiffuse, uv - off).r;
    54→      vec4 c = texture2D(tDiffuse, uv);
    55→      float b = texture2D(tDiffuse, uv + off).b;
    56→      vec3 col = vec3(r, c.g, b);
    57→      col += grain - 0.027;
    58→      // فينيت
    59→      float vig = smoothstep(0.95, 0.32, length(dir) * 1.32);
    60→      col *= mix(0.32, 1.0, vig);
    61→      // حواف حمراء عند الخطر/الجرح
    62→      float edge = smoothstep(0.55, 1.0, length(dir) * 1.6);
    63→      col += vec3(0.28, 0.015, 0.01) * edge * (uDamage * 1.4 + uThreat * 0.35);
    64→      col = mix(col, vec3(0.25, 0.02, 0.02), uDamage * 0.35 * edge);
    65→      gl_FragColor = vec4(col, c.a);
    66→    }
    67→  `,
    68→};

   149→  private initThree() {
   150→    const w = () => this.canvas.clientWidth || window.innerWidth;
   151→    const h = () => this.canvas.clientHeight || window.innerHeight;
   152→    this.renderer = new THREE.WebGLRenderer({
   153→      canvas: this.canvas,
   154→      antialias: true,
   155→      powerPreference: "high-performance",
   156→    });
   157→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   158→    this.renderer.setSize(w(), h(), false);
   159→    this.renderer.shadowMap.enabled = true;
   160→    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
   161→    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
   162→    this.renderer.toneMappingExposure = 1.05;
   163→
   164→    this.scene = new THREE.Scene();
   165→    this.scene.background = new THREE.Color(0x05070a);
   166→    this.scene.fog = new THREE.FogExp2(0x07080b, 0.021);
   167→
   168→    this.camera = new THREE.PerspectiveCamera(72, w() / h(), 0.08, 260);
   169→    this.scene.add(this.camera);
   170→
   171→    // إضاءة أساسية خافتة
   172→    const hemi = new THREE.HemisphereLight(0x27303a, 0x0a0a0c, 0.32);
   173→    this.scene.add(hemi);
   174→    const moon = new THREE.DirectionalLight(0x8a93a8, 0.22);
   175→    moon.position.set(-60, 90, -80);
   176→    this.scene.add(moon);
   177→
   178→    // كشاف
   179→    this.flashlight = new THREE.SpotLight(0xfff0d0, 0, 34, 0.5, 0.6, 1.5);
   180→    this.flashlight.castShadow = true;
   181→    this.flashlight.shadow.mapSize.set(1024, 1024);
   182→    this.flashlight.shadow.camera.near = 0.2;
   183→    this.flashlight.shadow.camera.far = 36;
   184→    this.flashlight.shadow.bias = -0.003;
   185→    this.flashlight.position.set(0.18, -0.14, 0);
   186→    this.camera.add(this.flashlight);
   187→    this.flashTarget = new THREE.Object3D();
   188→    this.flashTarget.position.set(0, -0.06, -6);
   189→    this.camera.add(this.flashTarget);
   190→    this.flashlight.target = this.flashTarget;
   191→
   192→    // نموذج السلاح
   193→    this.vmGroup = new THREE.Group();
   194→    this.vmGroup.position.set(0.26, -0.24, -0.5);
   195→    this.camera.add(this.vmGroup);
   196→    this.vmPistol = this.buildPistolVM();
   197→    this.vmShotgun = this.buildShotgunVM();
   198→    this.vmCrowbar = this.buildCrowbarVM();
   199→    this.vmGroup.add(this.vmPistol, this.vmShotgun, this.vmCrowbar);

   201→    this.muzzle.position.set(0.26, -0.1, -1.1);
   202→    this.camera.add(this.muzzle);
   203→
   204→    // رماد متساقط
   205→    const ashGeo = new THREE.BufferGeometry();
   206→    const N = 700;
   207→    const arr = new Float32Array(N * 3);
   208→    for (let i = 0; i < N; i++) {
   209→      arr[i * 3] = (Math.random() - 0.5) * 60;
   210→      arr[i * 3 + 1] = Math.random() * 26;
   211→      arr[i * 3 + 2] = (Math.random() - 0.5) * 60;
   212→    }
   213→    ashGeo.setAttribute("position", new THREE.BufferAttribute(arr, 3));
   214→    this.ash = new THREE.Points(
   215→      ashGeo,
   216→      new THREE.PointsMaterial({ color: 0x8a8478, size: 0.07, transparent: true, opacity: 0.5, sizeAttenuation: true }),

   219→

   224→    this.composer.addPass(bloom);
   225→    this.composer.addPass(new OutputPass());
   226→    this.horrorPass = new ShaderPass(HorrorShader);
   227→    this.composer.addPass(this.horrorPass);
   228→
   229→    window.addEventListener("resize", this.onResize);

   232→  private buildPistolVM(): THREE.Group {
   233→    const g = new THREE.Group();
   234→    const dark = new THREE.MeshStandardMaterial({ color: 0x23262a, roughness: 0.5, metalness: 0.7 });
   235→    const grip = new THREE.MeshStandardMaterial({ color: 0x1a1512, roughness: 0.85 });
   236→    const body = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.08, 0.26), dark);
   237→    body.position.z = -0.05;
   238→    const barrel = new THREE.Mesh(new THREE.BoxGeometry(0.03, 0.035, 0.16), dark);
   239→    barrel.position.set(0, 0.028, -0.2);
   240→    const handle = new THREE.Mesh(new THREE.BoxGeometry(0.045, 0.14, 0.06), grip);
   241→    handle.position.set(0, -0.09, 0.03);
   242→    handle.rotation.x = 0.22;
   243→    g.add(body, barrel, handle);
   244→    g.rotation.y = -0.06;
   245→    return g;
   246→  }

   250→    const wood = new THREE.MeshStandardMaterial({ color: 0x4a3220, roughness: 0.8 });
   251→    const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.024, 0.024, 0.62, 10), dark);
   252→    barrel.rotation.x = Math.PI / 2;
   253→    barrel.position.set(0, 0.02, -0.28);
   254→    const pump = new THREE.Mesh(new THREE.CylinderGeometry(0.032, 0.032, 0.16, 10), wood);
   255→    pump.rotation.x = Math.PI / 2;
   256→    pump.position.set(0, -0.03, -0.3);
   257→    const stock = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.09, 0.24), wood);
   258→    stock.position.set(0, -0.045, 0.1);
   259→    g.add(barrel, pump, stock);
   260→    g.rotation.y = -0.04;

   277→  // ══════════ input ══════════
   278→  private onKeyDown = (e: KeyboardEvent) => {
   279→    if (e.code === "Tab") e.preventDefault();
   280→    this.keys.add(e.code);
   281→    const st = useGame.getState();
   282→    if (st.screen === "playing") {
   283→      if (e.code === "KeyF") this.toggleFlashlight();
   284→      else if (e.code === "KeyR") this.startReload();
   285→      else if (e.code === "Digit1") this.switchWeapon("crowbar");
   286→      else if (e.code === "Digit2" && st.weapons.pistol) this.switchWeapon("pistol");
   287→      else if (e.code === "Digit3" && st.weapons.shotgun) this.switchWeapon("shotgun");
   288→      else if (e.code === "KeyE") this.tryInteract();
   289→      else if (e.code === "Tab") {
   290→        st.setScreen("inventory");
   291→        document.exitPointerLock();
   292→      } else if (e.code === "KeyM") {
   293→        st.setScreen("map");
   294→        document.exitPointerLock();
   295→      }
   296→    } else if (st.screen === "inventory" && (e.code === "Tab" || e.code === "Escape")) {
   297→      st.setScreen("playing");
   298→      this.lockPointer();
   299→    } else if (st.screen === "map" && (e.code === "KeyM" || e.code === "Escape" || e.code === "Tab")) {
   300→      st.setScreen("playing");
   301→      this.lockPointer();
   302→    }
   303→  };
   304→  private onKeyUp = (e: KeyboardEvent) => this.keys.delete(e.code);
   305→  private onMouseMove = (e: MouseEvent) => {
   306→    if (document.pointerLockElement !== this.canvas) return;
   307→    const st = useGame.getState();
   308→    if (st.screen !== "playing") return;
   309→    const s = 0.0022;
   310→    this.yaw -= e.movementX * s;
   311→    this.pitch -= e.movementY * s;
   312→    this.pitch = Math.max(-1.35, Math.min(1.35, this.pitch));
   313→  };
   314→  private onMouseDown = (e: MouseEvent) => {
   315→    if (e.button !== 0) return;
   316→    const st = useGame.getState();
   317→    if (st.screen !== "playing") return;
   318→    if (document.pointerLockElement !== this.canvas) {
   319→      this.lockPointer();
   320→      return;
   321→    }
   322→    this.mouseDown = true;
   323→    this.shoot();
   324→  };

   457→  // ══════════ run lifecycle ══════════
   458→  private beginRun() {
   459→    const st = useGame.getState();
   460→    st.resetRun();
   461→    if (this.pendingSave) {
   462→      st.applySave(this.pendingSave);
   463→      this.pendingSave = null;
   464→    }
   465→    if (!this.world) {
   466→      this.world = buildWorld(this.scene);
   467→      this.enemies = new EnemyManager(this.scene, {
   468→        damagePlayer: (n, fx, fz) => this.damagePlayer(n, fx, fz),
   469→        onDeath: (e) => this.onEnemyDeath(e),
   470→        spawnAdds: (x, z, kind, n) => {
   471→          for (let i = 0; i < n; i++) {
   472→            this.enemies!.addAt(x + (Math.random() - 0.5) * 6, z + 4 + i * 2, kind);
   473→          }
   474→        },
   475→      });

   598→  private shoot() {
   599→    const st = useGame.getState();
   600→    const w = st.hud.equipped;
   601→    if (!w || st.hud.reloading) return;
   602→    const def = WEAPONS[w];
   603→    const now = performance.now() / 1000;
   604→    if (now - this.lastShot < def.fireRate) return;
   605→
   606→    if (def.melee) {
   607→      this.lastShot = now;
   608→      this.swingT = 0.28;
   609→      this.vmRecoil = 1;
   610→      audio.play("swing");
   611→      // ضربة قريبة
   612→      this.raycaster.setFromCamera(new THREE.Vector2(0, 0), this.camera);
   613→      this.raycaster.far = def.range;
   614→      const hits = this.raycaster.intersectObjects(this.hitTargets(), true);
   615→      const hit = this.pickEnemyHit(hits);
   616→      if (hit) {
   617→        const e = hit.object.userData.enemy as Enemy;
   618→        this.enemies!.damageAt(e, def.damage, hit.object.userData.isHead === true);
   619→        this.spawnBlood(hit.point);
   620→        audio.play("hit_flesh", { volume: 0.8 });
   621→      }
   622→      return;
   623→    }
   624→
   625→    const mag = w === "pistol" ? st.hud.pistolMag : st.hud.shotgunMag;
   626→    if (mag <= 0) {
   627→      audio.play("dryfire");
   628→      this.startReload();
   629→      return;
   630→    }
   631→    this.lastShot = now;
   632→    st.addShot();
   633→    if (w === "pistol") st.setHud({ pistolMag: mag - 1 });
   634→    else st.setHud({ shotgunMag: mag - 1 });
   635→
   636→    audio.play(w === "pistol" ? "shot_pistol" : "shot_shotgun");
   637→    this.muzzleT = 0.06;
   638→    this.muzzle.intensity = 26;
   639→    this.vmRecoil = 1;
   640→    this.pitch = Math.min(1.35, this.pitch + (w === "shotgun" ? 0.035 : 0.016));
   641→    this.enemies!.noise(this.pos.x, this.pos.z, def.noise);
   642→
   643→    const pellets = def.pellets ?? 1;
   644→    for (let i = 0; i < pellets; i++) {
   645→      const spread = def.spread ?? 0.004;
   646→      const v = new THREE.Vector2(
   647→        (Math.random() - 0.5) * spread * 2,
   648→        (Math.random() - 0.5) * spread * 2,
   649→      );
   650→      this.raycaster.setFromCamera(v, this.camera);
   651→      this.raycaster.far = def.range;
   652→      const hits = this.raycaster.intersectObjects(this.hitTargets(), true);
   653→      const res = this.pickEnemyHit(hits);
   654→      if (res) {
   655→        const e = res.object.userData.enemy as Enemy;
   656→        const isHead = res.object.userData.isHead === true;
   657→        const dmg = def.damage * (isHead ? def.headMult : 1);
   658→        this.enemies!.damageAt(e, dmg, isHead);
   659→        this.spawnBlood(res.point);
   660→        audio.play(isHead ? "hit_head" : "hit_flesh", { volume: 0.7 });
   661→        if (isHead) st.addHeadshot();
   662→      } else if (hits.length > 0) {
   663→        // اصطدام بجدار — شرارة غبار
   664→        this.spawnBlood(hits[0].point, 0x6a655c, 3);
   665→      }
   666→    }
   667→  }

   802→  private findInteract(): Interactable | null {
   803→    if (!this.world) return null;
   804→    this.camera.getWorldDirection(this.forward);
   805→    let best: Interactable | null = null;
   806→    let bestScore = -1;
   807→    for (const it of this.world.interactables) {
   808→      if (it.used) continue;
   809→      const dx = it.x - this.pos.x;
   810→      const dz = it.z - this.pos.z;
   811→      const d = Math.hypot(dx, dz);
   812→      if (d > it.radius) continue;
   813→      const dot = (dx / (d || 1)) * this.forward.x + (dz / (d || 1)) * this.forward.z;
   814→      if (dot < 0.15 && d > 1.1) continue;
   815→      const score = dot + (it.radius - d);
   816→      if (score > bestScore) {
   817→        bestScore = score;
   818→        best = it;
   819→      }
   820→    }
   821→    return best;
   822→  }

   824→  private tryInteract() {

   845→        }

  1520→  private updateAudioDirectors(dt: number, st: ReturnType<typeof useGame.getState>) {
  1521→    void dt;
  1522→    if (!this.enemies) return;
  1523→    const { dist, chasing, boss } = this.enemies.nearestThreat(this.pos.x, this.pos.z);
  1524→    let threat = 0;
  1525→    if (chasing) threat = Math.max(0.35, 1 - dist / 30);
  1526→    if (boss && dist < 45) threat = Math.max(threat, 0.55);
  1527→    if (Math.abs(threat - st.hud.threat) > 0.05) {
  1528→      st.setHud({ threat });
  1529→      audio.setThreat(threat);
  1530→    }
  1531→    const wantHb = st.hud.hp < 30 || threat > 0.8;
  1532→    if (wantHb !== st.hud.heartbeat) {
  1533→      st.setHud({ heartbeat: wantHb });
  1534→      if (wantHb) audio.startHeartbeat();
  1535→      else audio.stopHeartbeat();
  1536→    }
  1537→    // تغيير الأجواء حسب المنطقة
  1538→    const x = this.pos.x;
  1539→    const z = this.pos.z;
  1540→    if (x > 560) {
  1541→      if (st.ambient !== "lab") {
  1542→        st.setAmbient("lab");
  1543→        audio.startAmbient("lab");
  1544→      }
  1545→    } else if (z < -92) {
  1546→      if (st.ambient !== "harbor") {
  1547→        st.setAmbient("harbor");
  1548→        audio.startAmbient("harbor");
  1549→      }
  1550→    } else if (st.ambient === "menu" || st.ambient === "lab" || st.ambient === "harbor") {
  1551→      st.setAmbient("city");
  1552→      audio.startAmbient("city");
  1553→    }
  1554→  }
  1555→}

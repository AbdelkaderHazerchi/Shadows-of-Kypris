    24→    24→const HorrorShader = {
    25→    25→  uniforms: {
    26→    26→    tDiffuse: { value: null as THREE.Texture | null },
    27→    27→    uTime: { value: 0 },
    28→    28→    uDamage: { value: 0 },
    29→    29→    uThreat: { value: 0 },
    30→    30→  },
    31→    31→  vertexShader: /* glsl */ `
    32→    32→    varying vec2 vUv;
    33→    33→    void main() {
    34→    34→      vUv = uv;
    35→    35→      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    36→    36→    }
    37→    37→  `,
    38→    38→  fragmentShader: /* glsl */ `
    39→    39→    uniform sampler2D tDiffuse;
    40→    40→    uniform float uTime;
    41→    41→    uniform float uDamage;
    42→    42→    uniform float uThreat;
    43→    43→    varying vec2 vUv;
    44→    44→    float rand(vec2 co) { return fract(sin(dot(co, vec2(12.9898, 78.233))) * 43758.5453); }
    45→    45→    void main() {
    46→    46→      vec2 uv = vUv;
    47→    47→      // تشويش خفيف
    48→    48→      float grain = rand(uv * vec2(1024.0 + mod(uTime * 60.0, 100.0), 768.0)) * 0.055;
    49→    49→      // انحراف لوني خفيف عند الحواف
    50→    50→      vec2 dir = uv - 0.5;
    51→    51→      float d2 = dot(dir, dir);
    52→    52→      vec2 off = dir * d2 * 0.012;
    53→    53→      float r = texture2D(tDiffuse, uv - off).r;
    54→    54→      vec4 c = texture2D(tDiffuse, uv);
    55→    55→      float b = texture2D(tDiffuse, uv + off).b;
    56→    56→      vec3 col = vec3(r, c.g, b);
    57→    57→      col += grain - 0.027;
    58→    58→      // فينيت
    59→    59→      float vig = smoothstep(0.95, 0.32, length(dir) * 1.32);
    60→    60→      col *= mix(0.32, 1.0, vig);
    61→    61→      // حواف حمراء عند الخطر/الجرح
    62→    62→      float edge = smoothstep(0.55, 1.0, length(dir) * 1.6);
    63→    63→      col += vec3(0.28, 0.015, 0.01) * edge * (uDamage * 1.4 + uThreat * 0.35);
    64→    64→      col = mix(col, vec3(0.25, 0.02, 0.02), uDamage * 0.35 * edge);
    65→    65→      gl_FragColor = vec4(col, c.a);
    66→    66→    }
    67→    67→  `,
    68→    68→};

   278→   278→  private onKeyDown = (e: KeyboardEvent) => {
   279→   279→    if (e.code === "Tab") e.preventDefault();
   280→   280→    this.keys.add(e.code);
   281→   281→    const st = useGame.getState();
   282→   282→    if (st.screen === "playing") {
   283→   283→      if (e.code === "KeyF") this.toggleFlashlight();
   284→   284→      else if (e.code === "KeyR") this.startReload();
   285→   285→      else if (e.code === "Digit1") this.switchWeapon("crowbar");
   286→   286→      else if (e.code === "Digit2" && st.weapons.pistol) this.switchWeapon("pistol");
   287→   287→      else if (e.code === "Digit3" && st.weapons.shotgun) this.switchWeapon("shotgun");
   288→   288→      else if (e.code === "KeyE") this.tryInteract();
   289→   289→      else if (e.code === "Tab") {
   290→   290→        st.setScreen("inventory");
   291→   291→        document.exitPointerLock();
   292→   292→      } else if (e.code === "KeyM") {
   293→   293→        st.setScreen("map");
   294→   294→        document.exitPointerLock();
   295→   295→      }
   296→   296→    } else if (st.screen === "inventory" && (e.code === "Tab" || e.code === "Escape")) {
   297→   297→      st.setScreen("playing");
   298→   298→      this.lockPointer();
   299→   299→    } else if (st.screen === "map" && (e.code === "KeyM" || e.code === "Escape" || e.code === "Tab")) {
   300→   300→      st.setScreen("playing");
   301→   301→      this.lockPointer();
   302→   302→    }

   598→   598→  private shoot() {
   599→   599→    const st = useGame.getState();
   600→   600→    const w = st.hud.equipped;
   601→   601→    if (!w || st.hud.reloading) return;
   602→   602→    const def = WEAPONS[w];
   603→   603→    const now = performance.now() / 1000;
   604→   604→    if (now - this.lastShot < def.fireRate) return;
   605→   605→
   606→   606→    if (def.melee) {
   607→   607→      this.lastShot = now;
   608→   608→      this.swingT = 0.28;
   609→   609→      this.vmRecoil = 1;
   610→   610→      audio.play("swing");
   611→   611→      // ضربة قريبة
   612→   612→      this.raycaster.setFromCamera(new THREE.Vector2(0, 0), this.camera);
   613→   613→      this.raycaster.far = def.range;
   614→   614→      const hits = this.raycaster.intersectObjects(this.hitTargets(), true);
   615→   615→      const hit = this.pickEnemyHit(hits);
   616→   616→      if (hit) {
   617→   617→        const e = hit.object.userData.enemy as Enemy;
   618→   618→        this.enemies!.damageAt(e, def.damage, hit.object.userData.isHead === true);
   619→   619→        this.spawnBlood(hit.point);
   620→   620→        audio.play("hit_flesh", { volume: 0.8 });
   621→   621→      }
   622→   622→      return;
   623→   623→    }
   624→   624→
   625→   625→    const mag = w === "pistol" ? st.hud.pistolMag : st.hud.shotgunMag;
   626→   626→    if (mag <= 0) {
   627→   627→      audio.play("dryfire");
   628→   628→      this.startReload();
   629→   629→      return;
   630→   630→    }
   631→   631→    this.lastShot = now;
   632→   632→    st.addShot();
   633→   633→    if (w === "pistol") st.setHud({ pistolMag: mag - 1 });
   634→   634→    else st.setHud({ shotgunMag: mag - 1 });
   635→   635→
   636→   636→    audio.play(w === "pistol" ? "shot_pistol" : "shot_shotgun");
   637→   637→    this.muzzleT = 0.06;
   638→   638→    this.muzzle.intensity = 26;
   639→   639→    this.vmRecoil = 1;
   640→   640→    this.pitch = Math.min(1.35, this.pitch + (w === "shotgun" ? 0.035 : 0.016));
   641→   641→    this.enemies!.noise(this.pos.x, this.pos.z, def.noise);
   642→   642→
   643→   643→    const pellets = def.pellets ?? 1;
   644→   644→    for (let i = 0; i < pellets; i++) {
   645→   645→      const spread = def.spread ?? 0.004;
   646→   646→      const v = new THREE.Vector2(
   647→   647→        (Math.random() - 0.5) * spread * 2,
   648→   648→        (Math.random() - 0.5) * spread * 2,
   649→   649→      );
   650→   650→      this.raycaster.setFromCamera(v, this.camera);
   651→   651→      this.raycaster.far = def.range;
   652→   652→      const hits = this.raycaster.intersectObjects(this.hitTargets(), true);
   653→   653→      const res = this.pickEnemyHit(hits);
   654→   654→      if (res) {
   655→   655→        const e = res.object.userData.enemy as Enemy;
   656→   656→        const isHead = res.object.userData.isHead === true;
   657→   657→        const dmg = def.damage * (isHead ? def.headMult : 1);
   658→   658→        this.enemies!.damageAt(e, dmg, isHead);
   659→   659→        this.spawnBlood(res.point);
   660→   660→        audio.play(isHead ? "hit_head" : "hit_flesh", { volume: 0.7 });
   661→   661→        if (isHead) st.addHeadshot();
   662→   662→      } else if (hits.length > 0) {
   663→   663→        // اصطدام بجدار — شرارة غبار
   664→   664→        this.spawnBlood(hits[0].point, 0x6a655c, 3);
   665→   665→      }
   666→   666→    }
   667→   667→  }

   798→   798→  // ══════════ interaction ══════════
   799→   799→  private currentInteract: Interactable | null = null;
   800→   800→  private forward = new THREE.Vector3();
   801→   801→
   802→   802→  private findInteract(): Interactable | null {
   803→   803→    if (!this.world) return null;
   804→   804→    this.camera.getWorldDirection(this.forward);
   805→   805→    let best: Interactable | null = null;
   806→   806→    let bestScore = -1;
   807→   807→    for (const it of this.world.interactables) {
   808→   808→      if (it.used) continue;
   809→   809→      const dx = it.x - this.pos.x;
   810→   810→      const dz = it.z - this.pos.z;
   811→   811→      const d = Math.hypot(dx, dz);
   812→   812→      if (d > it.radius) continue;
   813→   813→      const dot = (dx / (d || 1)) * this.forward.x + (dz / (d || 1)) * this.forward.z;
   814→   814→      if (dot < 0.15 && d > 1.1) continue;
   815→   815→      const score = dot + (it.radius - d);
   816→   816→      if (score > bestScore) {
   817→   817→        bestScore = score;
   818→   818→        best = it;
   819→   819→      }
   820→   820→    }
   821→   821→    return best;

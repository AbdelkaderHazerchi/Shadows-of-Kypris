   161→// ── Canvas texture helpers ──
   162→function canvasTex(
   163→  w: number,
   164→  h: number,
   165→  draw: (ctx: CanvasRenderingContext2D) => void,
   166→  repeat = 1,
   167→): THREE.Texture {
   168→  const c = document.createElement("canvas");
   169→  c.width = w;
   170→  c.height = h;
   171→  const ctx = c.getContext("2d")!;
   172→  draw(ctx);
   173→  const tex = new THREE.CanvasTexture(c);
   174→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
   175→  tex.repeat.set(repeat, repeat);
   176→  tex.anisotropy = 4;
   177→  return tex;
   178→}

   180→function noiseFill(
   181→  ctx: CanvasRenderingContext2D,
   182→  w: number,
   183→  h: number,
   184→  base: string,
   185→  amt: number,
   186→  dark: string,
   187→) {
   188→  ctx.fillStyle = base;
   189→  ctx.fillRect(0, 0, w, h);
   190→  const img = ctx.getImageData(0, 0, w, h);
   191→  const d = img.data;
   192→  for (let i = 0; i < d.length; i += 4) {
   193→    const n = (Math.random() - 0.5) * amt;
   194→    d[i] = Math.max(0, Math.min(255, d[i] + n));
   195→    d[i + 1] = Math.max(0, Math.min(255, d[i + 1] + n));
   196→    d[i + 2] = Math.max(0, Math.min(255, d[i + 2] + n));
   197→  }
   198→  ctx.putImageData(img, 0, 0);
   199→  for (let i = 0; i < 14; i++) {
   200→    const x = Math.random() * w;
   201→    const y = Math.random() * h;
   202→    const r = 6 + Math.random() * 30;
   203→    const g = ctx.createRadialGradient(x, y, 0, x, y, r);
   204→    g.addColorStop(0, dark);
   205→    g.addColorStop(1, "rgba(0,0,0,0)");
   206→    ctx.fillStyle = g;
   207→    ctx.fillRect(x - r, y - r, r * 2, r * 2);
   208→  }
   209→}

   380→function pushStatic(ctx: BuildCtx, bucket: string, geo: THREE.BufferGeometry, matrix: THREE.Matrix4) {
   381→  const g = geo.clone();
   382→  g.applyMatrix4(matrix);
   383→  if (!ctx.staticGeos.has(bucket)) ctx.staticGeos.set(bucket, []);
   384→  ctx.staticGeos.get(bucket)!.push(g);
   385→}

   387→function addBox(
   388→  ctx: BuildCtx,
   389→  bucket: string,
   390→  x: number,
   391→  y: number,
   392→  z: number,
   393→  w: number,
   394→  h: number,
   395→  d: number,
   396→  solid = true,
   397→) {
   398→  const m = new THREE.Matrix4().compose(
   399→    new THREE.Vector3(x, y + h / 2, z),
   400→    new THREE.Quaternion(),
   401→    new THREE.Vector3(w, h, d),
   402→  );
   403→  pushStatic(ctx, bucket, new THREE.BoxGeometry(1, 1, 1), m);
   404→  if (solid) {
   405→    ctx.colliders.push({
   406→      minX: x - w / 2,
   407→      maxX: x + w / 2,
   408→      minZ: z - d / 2,
   409→      maxZ: z + d / 2,
   410→    });
   411→  }
   412→}

   414→/** جدار مع فتحات للأبواب: gaps = [{at, width}] مواقع على طول الجدار */
   415→function wallWithGaps(
   416→  ctx: BuildCtx,
   417→  bucket: string,
   418→  x1: number,
   419→  z1: number,
   420→  x2: number,
   421→  z2: number,
   422→  gaps: { at: number; width: number }[] = [],
   423→) {
   424→  const horiz = Math.abs(x2 - x1) > Math.abs(z2 - z1);
   425→  const len = horiz ? Math.abs(x2 - x1) : Math.abs(z2 - z1);
   426→  const cx = (x1 + x2) / 2;
   427→  const cz = (z1 + z2) / 2;
   428→  const segs: { s: number; e: number }[] = [];
   429→  let cur = 0;
   430→  const sorted = [...gaps].sort((a, b) => a.at - b.at);
   431→  for (const g of sorted) {
   432→    const s = g.at - g.width / 2;
   433→    if (s > cur) segs.push({ s: cur, e: s });
   434→    cur = g.at + g.width / 2;
   435→  }
   436→  if (cur < len) segs.push({ s: cur, e: len });
   437→
   438→  for (const seg of segs) {
   439→    const segLen = seg.e - seg.s;
   440→    if (segLen <= 0.05) continue;
   441→    const mid = (seg.s + seg.e) / 2;
   442→    if (horiz) {
   443→      const sx = x1 + (x2 > x1 ? mid : -mid);
   444→      addBox(ctx, bucket, sx, 0, cz, segLen, WALL_H, T);
   445→    } else {
   446→      const sz = z1 + (z2 > z1 ? mid : -mid);
   447→      addBox(ctx, bucket, cx, 0, sz, T, WALL_H, segLen);
   448→    }
   449→  }
   450→  for (const g of sorted) {
   451→    const gapLen = g.width;
   452→    const mid = g.at;
   453→    if (horiz) {
   454→      const sx = x1 + (x2 > x1 ? mid : -mid);
   455→      addBox(ctx, bucket, sx, 2.6, cz, gapLen, WALL_H - 2.6, T, false);
   456→    } else {
   457→      const sz = z1 + (z2 > z1 ? mid : -mid);
   458→      addBox(ctx, bucket, cx, 2.6, sz, T, WALL_H - 2.6, gapLen, false);
   459→    }
   460→  }
   461→}

   496→function addFire(ctx: BuildCtx, x: number, z: number, scale = 1) {
   497→  const l = new THREE.PointLight(0xd4562a, 3.4 * scale, 16 * scale, 1.7);
   498→  l.position.set(x, 1 + scale, z);
   499→  ctx.scene.add(l);
   500→  ctx.fireLights.push({ light: l, base: 3.4 * scale });
   501→  const p = propCtxOf(ctx);
   502→  // لهب مخروطي مزدوج
   503→  const flame = new THREE.ConeGeometry(0.42 * scale, 1.5 * scale, 7);
   504→  flame.translate(x, 0.75 * scale, z);
   505→  p.push("fireCone", flame, mat4(0, 0, 0));
   506→  const core = new THREE.ConeGeometry(0.24 * scale, 1.0 * scale, 6);
   507→  core.translate(x, 0.6 * scale, z);
   508→  p.push("fireCore", core, mat4(0, 0, 0));
   509→  // دخان
   510→  for (let i = 0; i < 3; i++) {
   511→    const sm = new THREE.SphereGeometry((0.5 + i * 0.35) * scale, 7, 6);
   512→    sm.translate(x + (Math.random() - 0.5) * 0.4, (1.8 + i * 0.9) * scale, z);
   513→    p.push("smoke", sm, mat4(0, 0, 0));
   514→  }
   515→  // أثر الحرق
   516→  const decal = new THREE.Mesh(
   517→    new THREE.CircleGeometry(1.6 * scale, 12),
   518→    new THREE.MeshBasicMaterial({ color: 0x090909 }),
   519→  );
   520→  decal.rotation.x = -Math.PI / 2;
   521→  decal.position.set(x, 0.03, z);
   522→  ctx.scene.add(decal);
   523→  decal.userData.noHit = true;
   524→}

   536→function bloodTrail(ctx: BuildCtx, x1: number, z1: number, x2: number, z2: number) {

   539→    const t = i / steps;
   540→    bloodDecal(ctx, x1 + (x2 - x1) * t + (Math.random() - 0.5) * 0.3, z1 + (z2 - z1) * t + (Math.random() - 0.5) * 0.3, 0.5 + Math.random() * 0.5);
   541→  }
   542→}

   596→function addDoorAt(
   597→  ctx: BuildCtx,
   598→  id: string,
   599→  cx: number,
   600→  cz: number,
   601→  orient: "x" | "z",
   602→  width: number,
   603→  style: "wood" | "metal" | "double" = "wood",

   606→  if (orient === "x") {
   607→    def.group.position.set(cx - width / 2, 0, cz);
   608→    def.group.rotation.y = 0;
   609→  } else {
   610→    def.group.position.set(cx, 0, cz - width / 2);
   611→    def.group.rotation.y = -Math.PI / 2;
   612→  }
   613→  def.group.traverse((o) => {
   614→    if (o instanceof THREE.Mesh) o.userData.doorMesh = id;
   615→  });
   616→  ctx.scene.add(def.group);
   617→  const half = width / 2 + 0.08;
   618→  const collider: Collider =
   619→    orient === "x"
   620→      ? { minX: cx - half, maxX: cx + half, minZ: cz - 0.14, maxZ: cz + 0.14 }
   621→      : { minX: cx - 0.14, maxX: cx + 0.14, minZ: cz - half, maxZ: cz + half };
   622→  ctx.doors.push({ id, def, collider, open: false });
   623→  ctx.interactables.push({
   624→    id,
   625→    kind: "door",
   626→    x: cx,
   627→    z: cz,
   628→    radius: 2.3,
   629→    prompt: "فتح / إغلاق الباب",
   630→    data: { doorId: id },
   631→    used: false,
   632→  });
   633→}

  1613→function debrisPile(ctx: BuildCtx, x: number, z: number, w2: number, d: number) {
  1614→  const rnd2 = seededRandom(Math.floor((x * 31 + z * 17) | 0) || 7);
  1615→  const rndLocal = (a: number, b: number) => a + rnd2() * (b - a);
  1616→  for (let i = 0; i < Math.floor((w2 * d) / 6); i++) {
  1617→    const bw = rndLocal(1.2, 3.6);
  1618→    const bh = rndLocal(0.6, 2.4);
  1619→    const bd = rndLocal(1.2, 3.4);
  1620→    const m = new THREE.Matrix4().compose(
  1621→      new THREE.Vector3(x + rndLocal(-w2 / 2.2, w2 / 2.2), bh / 2 + rndLocal(0, 0.8), z + rndLocal(-d / 2.2, d / 2.2)),
  1622→      new THREE.Quaternion().setFromEuler(new THREE.Euler(rndLocal(-0.2, 0.2), rndLocal(0, Math.PI), rndLocal(-0.2, 0.2))),
  1623→      new THREE.Vector3(bw, bh, bd),
  1624→    );
  1625→    pushStatic(ctx, "rubble", new THREE.BoxGeometry(1, 1, 1), m);
  1626→  }
  1627→  ctx.colliders.push({ minX: x - w2 / 2, maxX: x + w2 / 2, minZ: z - d / 2, maxZ: z + d / 2 });

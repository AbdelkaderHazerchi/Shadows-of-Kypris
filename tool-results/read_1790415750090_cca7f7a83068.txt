    44→class AudioEngine {
    45→  ctx: AudioContext | null = null;
    46→  private master!: GainNode;
    47→  private musicBus!: GainNode;
    48→  private sfxBus!: GainNode;
    49→  private ambBus!: GainNode;
    50→  private noiseBuf: AudioBuffer | null = null;
    51→  private ambNodes: { stop: () => void }[] = [];
    52→  private ambTimer: ReturnType<typeof setTimeout> | null = null;
    53→  private currentAmbient: AmbientProfile | null = null;
    54→  private threatGains: GainNode[] = [];
    55→  private hbTimer: ReturnType<typeof setInterval> | null = null;
    56→  private volumes = { master: 0.8, music: 0.7, sfx: 0.9 };
    57→  private started = false;
    58→
    59→  init() {
    60→    if (this.started) {
    61→      this.ctx?.resume();
    62→      return;
    63→    }
    64→    const AC = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    65→    if (!AC) return;
    66→    this.ctx = new AC();
    67→    const ctx = this.ctx;
    68→    this.master = ctx.createGain();
    69→    this.master.gain.value = this.volumes.master;
    70→    this.master.connect(ctx.destination);
    71→    this.musicBus = ctx.createGain();
    72→    this.musicBus.gain.value = this.volumes.music * 0.9;
    73→    this.musicBus.connect(this.master);
    74→    this.sfxBus = ctx.createGain();
    75→    this.sfxBus.gain.value = this.volumes.sfx;
    76→    this.sfxBus.connect(this.master);
    77→    this.ambBus = ctx.createGain();
    78→    this.ambBus.gain.value = this.volumes.music;
    79→    this.ambBus.connect(this.master);
    80→
    81→    // noise buffer 2s
    82→    const len = ctx.sampleRate * 2;
    83→    const buf = ctx.createBuffer(1, len, ctx.sampleRate);
    84→    const data = buf.getChannelData(0);
    85→    let last = 0;
    86→    for (let i = 0; i < len; i++) {
    87→      const white = Math.random() * 2 - 1;
    88→      last = (last + 0.02 * white) / 1.02; // brownish
    89→      data[i] = white * 0.4 + last * 3.2;
    90→    }
    91→    this.noiseBuf = buf;
    92→    this.started = true;
    93→  }
    94→
    95→  setVolumes(v: { master?: number; music?: number; sfx?: number }) {
    96→    Object.assign(this.volumes, v);
    97→    if (!this.ctx) return;
    98→    const t = this.ctx.currentTime;
    99→    this.master.gain.setTargetAtTime(this.volumes.master, t, 0.05);
   100→    this.musicBus.gain.setTargetAtTime(this.volumes.music * 0.9, t, 0.05);
   101→    this.ambBus.gain.setTargetAtTime(this.volumes.music, t, 0.05);
   102→    this.sfxBus.gain.setTargetAtTime(this.volumes.sfx, t, 0.05);
   103→  }
   104→
   105→  private out(bus?: GainNode) {
   106→    return bus ?? this.sfxBus;
   107→  }
   108→
   109→  private noiseSource(dur: number, rate = 1): AudioBufferSourceNode | null {
   110→    if (!this.ctx || !this.noiseBuf) return null;
   111→    const src = this.ctx.createBufferSource();
   112→    src.buffer = this.noiseBuf;
   113→    src.loop = true;
   114→    src.playbackRate.value = rate;
   115→    return src;
   116→  }
   117→
   118→  private env(g: GainNode, t0: number, a: number, peak: number, d: number) {
   119→    g.gain.setValueAtTime(0.0001, t0);
   120→    g.gain.exponentialRampToValueAtTime(Math.max(peak, 0.0001), t0 + a);
   121→    g.gain.exponentialRampToValueAtTime(0.0001, t0 + a + d);
   122→  }
   123→
   124→  private panner(pan: number): StereoPannerNode | null {
   125→    if (!this.ctx) return null;
   126→    const p = this.ctx.createStereoPanner();
   127→    p.pan.value = Math.max(-1, Math.min(1, pan));
   128→    return p;
   129→  }

   131→  // ── public sfx ──
   132→  play(name: SfxName, opts: SfxOpts = {}) {

   135→    const t0 = ctx.currentTime + 0.01;

   137→    const rate = opts.rate ?? 1;

   140→      const p = this.panner(opts.pan);
   141→      if (p) {
   142→        p.connect(this.out());
   143→        dest = p;
   144→      }

   154→        try {
   155→          r.nodes.forEach((n) => (n as OscillatorNode & AudioBufferSourceNode).stop?.());

   157→          /* noop */
   158→        }
   159→      }, (r.dur + 0.5) * 1000);

   176→          src.start(t0);
   177→          src.stop(t0 + 0.25);
   178→          return { nodes: [src], dur: 0.25 };
   179→        });
   180→        break;
   181→      }

   193→          src.start(t0);
   194→          src.stop(t0 + 0.4);

   201→          o.connect(og);
   202→          og.connect(dest);
   203→          o.start(t0);
   204→          o.stop(t0 + 0.25);

   206→        });

   685→  // ── ambient beds ──
   686→  startAmbient(profile: AmbientProfile) {
   687→    if (!this.ctx || !this.started) return;
   688→    if (this.currentAmbient === profile) return;
   689→    this.stopAmbient();
   690→    this.currentAmbient = profile;
   691→    const ctx = this.ctx;
   692→
   693→    const mkDrone = (freqs: number[], gainV: number, lfoF: number, type: OscillatorType = "sine") => {
   694→      const g = ctx.createGain();
   695→      g.gain.value = gainV;
   696→      g.connect(this.ambBus);
   697→      const nodes: (OscillatorNode | AudioBufferSourceNode)[] = [];
   698→      freqs.forEach((f, i) => {
   699→        const o = ctx.createOscillator();
   700→        o.type = type;
   701→        o.frequency.value = f * (1 + i * 0.003);
   702→        const og = ctx.createGain();
   703→        og.gain.value = 1 / freqs.length;
   704→        o.connect(og);
   705→        og.connect(g);
   706→        o.start();
   707→        nodes.push(o);
   708→      });
   709→      const lfo = ctx.createOscillator();
   710→      lfo.frequency.value = lfoF;
   711→      const lg = ctx.createGain();
   712→      lg.gain.value = gainV * 0.4;
   713→      lfo.connect(lg);
   714→      lg.connect(g.gain);
   715→      lfo.start();
   716→      nodes.push(lfo);
   717→      nodes.forEach((n) => this.ambNodes.push({ stop: () => n.stop() }));
   718→      return { nodes, g };
   719→    };
   720→
   721→    const mkWind = (gainV: number, cutoff: number, rate: number) => {
   722→      if (!this.noiseBuf) return { nodes: [] as AudioBufferSourceNode[], g: ctx.createGain() };
   723→      const src = ctx.createBufferSource();
   724→      src.buffer = this.noiseBuf;
   725→      src.loop = true;
   726→      src.playbackRate.value = rate;
   727→      const f = ctx.createBiquadFilter();
   728→      f.type = "lowpass";
   729→      f.frequency.value = cutoff;
   730→      const g = ctx.createGain();
   731→      g.gain.value = gainV;
   732→      const lfo = ctx.createOscillator();
   733→      lfo.frequency.value = 0.07 + Math.random() * 0.05;
   734→      const lg = ctx.createGain();
   735→      lg.gain.value = gainV * 0.55;
   736→      lfo.connect(lg);
   737→      lg.connect(g.gain);
   738→      src.connect(f);
   739→      f.connect(g);
   740→      g.connect(this.ambBus);
   741→      src.start();
   742→      lfo.start();
   743→      ([src, lfo] as (AudioBufferSourceNode | OscillatorNode)[]).forEach((n) =>
   744→        this.ambNodes.push({ stop: () => n.stop() }),
   745→      );
   746→      return { nodes: [src, lfo] as (AudioBufferSourceNode | OscillatorNode)[], g };
   747→    };
   748→
   749→    // threat layer (persistent, gain 0)
   750→    const tg = ctx.createGain();
   751→    tg.gain.value = 0;
   752→    tg.connect(this.musicBus);
   753→    [110, 116.54, 155.56].forEach((f) => {
   754→      const o = ctx.createOscillator();
   755→      o.type = "sawtooth";
   756→      o.frequency.value = f;
   757→      const og = ctx.createGain();
   758→      og.gain.value = 0.33;
   759→      o.connect(og);
   760→      og.connect(tg);
   761→      o.start();
   762→      this.ambNodes.push({ stop: () => o.stop() });
   763→    });
   764→    const pulse = ctx.createOscillator();
   765→    pulse.type = "sine";
   766→    pulse.frequency.value = 2.2;
   767→    const pg = ctx.createGain();
   768→    pg.gain.value = 0.25;
   769→    pulse.connect(pg);
   770→    pg.connect(tg.gain);
   771→    pulse.start();
   772→    this.ambNodes.push({ stop: () => pulse.stop() });
   773→    this.threatGains = [tg];
   774→
   775→    const scheduleCreaks = (minGap: number, maxGap: number, sfx: SfxName, vol: number) => {
   776→      const loop = () => {
   777→        this.ambTimer = setTimeout(
   778→          () => {
   779→            if (this.currentAmbient === profile) this.play(sfx, { volume: vol, pan: Math.random() * 1.6 - 0.8 });
   780→            loop();
   781→          },
   782→          minGap + Math.random() * (maxGap - minGap),
   783→        );
   784→      };
   785→      loop();
   786→    };
   787→
   788→    switch (profile) {
   789→      case "menu":
   790→        mkDrone([55, 58.27], 0.12, 0.05);
   791→        mkWind(0.1, 300, 0.4);
   792→        scheduleCreaks(5000, 11000, "metal_creak", 0.5);
   793→        break;
   794→      case "city":
   795→        mkDrone([49, 51.9], 0.1, 0.04);
   796→        mkWind(0.16, 420, 0.5);
   797→        mkDrone([98], 0.05, 0.03, "triangle");
   798→        scheduleCreaks(6000, 14000, "metal_creak", 0.7);
   799→        scheduleCreaks(9000, 20000, "growl_far", 1);
   800→        scheduleCreaks(15000, 30000, "thunder", 0.7);
   801→        break;
   802→      case "indoor":
   803→        mkDrone([62, 65.4], 0.09, 0.06);
   804→        mkWind(0.07, 200, 0.35);
   805→        scheduleCreaks(7000, 15000, "metal_creak", 0.8);
   806→        scheduleCreaks(8000, 18000, "growl_far", 0.9);
   807→        break;
   808→      case "lab": {
   809→        mkDrone([60, 120.5], 0.11, 0.08, "square");
   810→        mkWind(0.06, 180, 0.3);
   811→        const beepLoop = () => {
   812→          this.ambTimer = setTimeout(() => {
   813→            if (this.currentAmbient === "lab" && this.ctx) {
   814→              const t0 = this.ctx.currentTime + 0.01;
   815→              const o = this.ctx.createOscillator();
   816→              o.type = "sine";
   817→              o.frequency.value = 1400 + Math.random() * 1200;
   818→              const g = this.ctx.createGain();
   819→              this.env(g, t0, 0.01, 0.05, 0.2);
   820→              o.connect(g);
   821→              g.connect(this.ambBus);
   822→              o.start(t0);
   823→              o.stop(t0 + 0.3);
   824→            }
   825→            beepLoop();
   826→          }, 3000 + Math.random() * 7000);
   827→        };
   828→        beepLoop();
   829→        scheduleCreaks(9000, 18000, "growl_far", 1);
   830→        break;
   831→      }
   832→      case "harbor":
   833→        mkDrone([45], 0.09, 0.03);
   834→        mkWind(0.22, 700, 0.6);
   835→        scheduleCreaks(5000, 10000, "metal_creak", 0.9);
   836→        break;
   837→    }
   838→  }

   841→    if (!this.ctx || this.threatGains.length === 0) return;
   842→    const v = Math.max(0, Math.min(1, t));

   845→    );

   849→    this.ambNodes.forEach((n) => {
   850→      try {
   851→        n.stop();
   852→      } catch {
   853→        /* noop */
   854→      }

   856→    this.ambNodes = [];
   857→    this.threatGains = [];

   859→      clearTimeout(this.ambTimer);
   860→      this.ambTimer = null;
   861→    }
   862→    this.currentAmbient = null;
   863→  }

   868→  playVoice(name: string, onEnd?: () => void): boolean {
   869→    const a = new Audio(`/audio/voices/${name}.mp3`);
   870→    a.volume = this.volumes.master;

   873→      ok = false;
   874→      onEnd?.();
   875→    };
   876→    a.onended = () => onEnd?.();
   877→    a.play().catch(() => {
   878→      ok = false;
   879→      onEnd?.();
   880→    });
   881→    this.currentVoice = a;
   882→    return ok;
   883→  }

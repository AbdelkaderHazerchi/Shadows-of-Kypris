    24→export interface Mat4Opts {
    25→  rx?: number;
    26→  ry?: number;
    27→  rz?: number;

    33→const _m4 = new THREE.Matrix4();
    34→const _q = new THREE.Quaternion();
    35→const _e = new THREE.Euler();
    36→const _v = new THREE.Vector3();
    37→const _s = new THREE.Vector3();
    38→
    39→/** مصفوفة سريعة: موضع + دوران + مقياس */
    40→export function mat4(x: number, y: number, z: number, o: Mat4Opts = {}): THREE.Matrix4 {
    41→  _e.set(o.rx ?? 0, o.ry ?? 0, o.rz ?? 0);
    42→  _q.setFromEuler(_e);
    43→  _v.set(x, y, z);
    44→  _s.set(o.sx ?? 1, o.sy ?? 1, o.sz ?? 1);
    45→  return _m4.compose(_v, _q, _s);
    46→}

    48→/** مولّد أرقام شبه عشوائي ثابت البذرة — للمدينة نفسها كل مرة */
    49→export function seededRandom(seed: number) {
    50→  let s = seed >>> 0;
    51→  return () => {
    52→    s = (s * 1664525 + 1013904223) >>> 0;
    53→    return s / 4294967296;
    54→  };
    55→}

    62→function noiseCanvasTex(
    63→  size: number,
    64→  base: string,
    65→  amt: number,
    66→  draw?: (ctx: CanvasRenderingContext2D, s: number) => void,
    67→  repeat = 2,
    68→): THREE.Texture {
    69→  const c = document.createElement("canvas");
    70→  c.width = c.height = size;
    71→  const ctx = c.getContext("2d")!;
    72→  ctx.fillStyle = base;
    73→  ctx.fillRect(0, 0, size, size);
    74→  const img = ctx.getImageData(0, 0, size, size);
    75→  const d = img.data;
    76→  for (let i = 0; i < d.length; i += 4) {
    77→    const n = (Math.random() - 0.5) * amt;
    78→    d[i] = Math.max(0, Math.min(255, d[i] + n));
    79→    d[i + 1] = Math.max(0, Math.min(255, d[i + 1] + n));
    80→    d[i + 2] = Math.max(0, Math.min(255, d[i + 2] + n));
    81→  }
    82→  ctx.putImageData(img, 0, 0);
    83→  draw?.(ctx, size);
    84→  const tex = new THREE.CanvasTexture(c);
    85→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    86→  tex.repeat.set(repeat, repeat);
    87→  tex.anisotropy = 4;
    88→  return tex;
    89→}

   245→class PB {
   246→  constructor(private ctx: PropCtx, private ox = 0, private oy = 0, private oz = 0) {}
   247→  /** إحداثيات محلية → عالمية */
   248→  private toWorld(x: number, y: number, z: number, o: Mat4Opts = {}) {
   249→    // دوران المجموعة حول Y فقط (كل الدعائم محاذاة أو مُدارة من العالم)
   250→    const c = Math.cos(this.grpRy), s = Math.sin(this.grpRy);
   251→    return {
   252→      x: this.ox + x * c + z * s,
   253→      y: this.oy + y,
   254→      z: this.oz - x * s + z * c,
   255→      ry: (o.ry ?? 0) + this.grpRy,
   256→      rx: o.rx ?? 0,
   257→      rz: o.rz ?? 0,
   258→      sx: o.sx ?? 1, sy: o.sy ?? 1, sz: o.sz ?? 1,
   259→    };
   260→  }
   261→  private grpRy = 0;
   262→  group(x: number, y: number, z: number, ry = 0) { this.ox = x; this.oy = y; this.oz = z; this.grpRy = ry; return this; }
   263→  box(bucket: string, x: number, y: number, z: number, w: number, h: number, d: number, o: Mat4Opts = {}) {
   264→    const p = this.toWorld(x, y, z, o);
   265→    this.ctx.push(bucket, BOX, mat4(p.x, p.y, p.z, { ...o, ry: p.ry, rx: p.rx, rz: p.rz, sx: w * p.sx, sy: h * p.sy, sz: d * p.sz }));
   266→    return this;
   267→  }
   268→  /** صندوق بمركز القاع (y = قاعدة) */
   269→  boxB(bucket: string, x: number, y: number, z: number, w: number, h: number, d: number, o: Mat4Opts = {}) {
   270→    return this.box(bucket, x, y + h / 2, z, w, h, d, o);
   271→  }
   272→  cyl(bucket: string, x: number, y: number, z: number, rTop: number, rBot: number, h: number, o: Mat4Opts & { seg?: number } = {}) {
   273→    const p = this.toWorld(x, y, z, o);
   274→    const g = new THREE.CylinderGeometry(rTop, rBot, h, o.seg ?? 10, 1);
   275→    if (o.rx || o.ry || o.rz) g.rotateX(o.rx ?? 0), g.rotateY(o.ry ?? 0), g.rotateZ(o.rz ?? 0);
   276→    g.translate(p.x, p.y, p.z);
   277→    this.ctx.push(bucket, g, mat4(0, 0, 0));
   278→    return this;
   279→  }
   280→  sph(bucket: string, x: number, y: number, z: number, r: number, o: Mat4Opts = {}) {
   281→    const p = this.toWorld(x, y, z, o);
   282→    this.ctx.push(bucket, UNIT_SPHERE, mat4(p.x, p.y, p.z, { ...o, ry: p.ry, sx: r * p.sx, sy: r * p.sy, sz: r * p.sz }));
   283→    return this;
   284→  }
   285→  /** استنساخ هندسة جاهزة بمصفوفة (للأنابيب المعقدة) */
   286→  geo(bucket: string, geo: THREE.BufferGeometry, x: number, y: number, z: number, o: Mat4Opts = {}) {
   287→    const p = this.toWorld(x, y, z, o);
   288→    const g = geo.clone();
   289→    if (o.rx || o.ry || o.rz) g.rotateX(o.rx ?? 0), g.rotateY(o.ry ?? 0), g.rotateZ(o.rz ?? 0);
   290→    g.translate(p.x, p.y, p.z);
   291→    this.ctx.push(bucket, g, mat4(0, 0, 0));
   292→    return this;
   293→  }
   294→  /** مصادم AABB بمقاسات محلية (غير مُدار) */
   295→  hit(x: number, z: number, w: number, d: number) {
   296→    const c = Math.cos(this.grpRy), s = Math.sin(this.grpRy);
   297→    const wx = this.ox + x * c + z * s;
   298→    const wz = this.oz - x * s + z * c;
   299→    const ew = Math.abs(c) * w / 2 + Math.abs(s) * d / 2;
   300→    const ed = Math.abs(s) * w / 2 + Math.abs(c) * d / 2;
   301→    this.ctx.collider(wx - ew, wx + ew, wz - ed, wz + ed);
   302→  }
   303→}

   307→export type CarVariant = "sedan" | "wreck" | "taxi" | "police" | "ambulance" | "van" | "truck";
   308→
   309→/**
   310→ * سيارة مفصلة: هيكل + كابينة زجاج + عجلات + مصدّات + مصابيح.
   311→ * الطول على محور Z المحلي (المقدمة نحو +Z بعد الدوران ry).
   312→ */
   313→export function car(w: PropCtx, x: number, z: number, ry = 0, variant: CarVariant = "sedan", seed = Math.random()) {
   314→  const p = new PB(w).group(x, 0, z, ry);
   315→  const rnd = seededRandom(Math.floor(seed * 1e9));
   316→  const bodyBucket =
   317→    variant === "police" ? "carPolice" :
   318→    variant === "taxi" ? "carTaxi" :
   319→    variant === "ambulance" ? "ambulance" :
   320→    variant === "wreck" ? "carBurned" :
   321→    variant === "truck" ? "carB" :
   322→    ["carA", "carB", "carC"][Math.floor(rnd() * 3)];
   323→  const burned = variant === "wreck";
   324→
   325→  if (variant === "truck") {
   326→    // شاحنة عسكرية (الميناء)
   327→    p.boxB(bodyBucket, 0, 0.55, -1.2, 2.3, 1.5, 4.6);           // كابينة+حمصة
   328→    p.boxB(bodyBucket, 0, 0.75, 2.2, 2.4, 1.9, 3.6);            // صندوق حمولة
   329→    p.boxB("canvasTop", 0, 2.65, 2.2, 2.3, 0.1, 3.4);           // غطاء قماشي
   330→    p.boxB("chrome", 0, 0.3, 3.9, 2.2, 0.35, 0.2);              // مصدّ أمامي
   331→    for (const [wx, wz] of [[-1.15, 2.9], [1.15, 2.9], [-1.15, -1.5], [1.15, -1.5], [-1.15, -2.6], [1.15, -2.6]] as const) {
   332→      p.cyl("rubber", wx, 0.52, wz, 0.52, 0.52, 0.34, { rx: Math.PI / 2 });
   333→    }
   334→    p.hit(0, 0, 3.2, 8.4);
   335→    return;
   336→  }
   337→
   338→  const L = variant === "van" || variant === "ambulance" ? 5.2 : 4.5;
   339→  const W = variant === "van" || variant === "ambulance" ? 2.1 : 1.86;
   340→  const wheelR = 0.34;
   341→
   342→  // هيكل سفلي + غطاء محرك وصندوق
   343→  p.boxB(bodyBucket, 0, wheelR * 0.55, 0, W, 0.55, L);
   344→  if (variant === "van" || variant === "ambulance") {
   345→    p.boxB(bodyBucket, 0, wheelR * 0.55 + 0.55, -0.15, W, 1.05, L - 0.5);
   346→    p.boxB("glassDark", 0, wheelR * 0.55 + 1.15, 1.6, W - 0.3, 0.5, 0.1);
   347→  } else {
   348→    p.boxB(bodyBucket, 0, wheelR * 0.55 + 0.55, 0.6, W - 0.15, 0.32, L * 0.32); // غطاء محرك منخفض
   349→    // كابينة
   350→    p.boxB(bodyBucket, 0, wheelR * 0.55 + 0.55, -0.45, W - 0.14, 0.62, L * 0.42);
   351→    // زجاج أمامي/خلفي وجوانب
   352→    p.boxB("glassDark", 0, wheelR * 0.55 + 0.95, -0.45 + L * 0.21, W - 0.2, 0.44, 0.06, { rx: -0.28 });
   353→    p.boxB("glassDark", 0, wheelR * 0.55 + 0.95, -0.45 - L * 0.21, W - 0.2, 0.4, 0.06, { rx: 0.32 });
   354→    p.boxB("glassDark", -(W - 0.12) / 2, wheelR * 0.55 + 0.92, -0.45, 0.05, 0.4, L * 0.36);

   531→export function clockTower(w: PropCtx, x: number, z: number): { hands: THREE.Object3D[] } {
   532→  const p = new PB(w).group(x, 0, z, 0);
   533→  const H = 21;
   534→  p.boxB("brick", 0, 0, 0, 6.4, H, 6.4);
   535→  p.boxB("concrete", 0, H, 0, 7.2, 0.6, 7.2);
   536→  p.boxB("brick", 0, H + 0.6, 0, 4.6, 3.2, 4.6);
   537→  p.boxB("concrete", 0, H + 3.8, 0, 5.4, 0.35, 5.4);
   538→  // هرم القمة
   539→  const cone = new THREE.ConeGeometry(3.4, 3.2, 4);
   540→  cone.rotateY(Math.PI / 4);
   541→  cone.translate(x, H + 5.55, z);
   542→  w.push("roof", cone, mat4(0, 0, 0));
   543→  // أعمدة زخرفية في القاعدة
   544→  for (const [ox, oz] of [[-2.9, -2.9], [2.9, -2.9], [-2.9, 2.9], [2.9, 2.9]] as const) {
   545→    p.boxB("concrete", ox, 0, oz, 0.7, 4.4, 0.7);
   546→  }
   547→  p.hit(0, 0, 6.8, 6.8);
   548→
   549→  // أوجه الساعة الأربعة (شرائح منفصلة قابلة للدوران)
   550→  const hands: THREE.Object3D[] = [];
   551→  const faceMat = new THREE.MeshStandardMaterial({ color: 0x1a1712, roughness: 0.85 });
   552→  const handMat = new THREE.MeshStandardMaterial({ color: 0xb8a86a, emissive: 0x6a5a20, emissiveIntensity: 0.7, roughness: 0.4 });
   553→  for (let i = 0; i < 4; i++) {
   554→    const faceG = new THREE.Group();
   555→    const ry = (i * Math.PI) / 2;
   556→    faceG.rotation.y = ry;
   557→    faceG.position.set(x + Math.sin(ry) * 3.25, H + 1.9, z + Math.cos(ry) * 3.25);
   558→    const face = new THREE.Mesh(new THREE.CircleGeometry(1.05, 24), faceMat);
   559→    faceG.add(face);
   560→    const hourH = new THREE.Mesh(new THREE.BoxGeometry(0.09, 0.52, 0.03), handMat);
   561→    hourH.position.y = 0.2;
   562→    const hourPivot = new THREE.Group();
   563→    hourPivot.add(hourH);
   564→    const minH = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.82, 0.03), handMat);
   565→    minH.position.y = 0.34;
   566→    const minPivot = new THREE.Group();
   567→    minPivot.add(minH);
   568→    faceG.add(hourPivot, minPivot);
   569→    faceG.userData.isProp = true;
   570→    hands.push(hourPivot, minPivot);
   571→    faceG.traverse((o) => { if (o instanceof THREE.Mesh) o.userData.noHit = true; });
   572→  }
   573→  return { hands };
   574→}

   936→/** نموذج ناجٍ — جسم مفصلي بوضعية تعبّ عن حالته */
   937→export function survivorModel(id: string): THREE.Group {
   938→  const pose = SURVIVOR_POSES[id] ?? SURVIVOR_POSES.adel;
   939→  const g = new THREE.Group();
   940→  const skin = new THREE.MeshStandardMaterial({ color: pose.skin, roughness: 0.9 });
   941→  const shirt = new THREE.MeshStandardMaterial({ color: pose.shirt, roughness: 0.95 });
   942→  const pants = new THREE.MeshStandardMaterial({ color: pose.pants, roughness: 0.95 });
   943→  const hair = new THREE.MeshStandardMaterial({ color: pose.hair, roughness: 1 });
   944→
   945→  const mkMesh = (geo: THREE.BufferGeometry, mat: THREE.Material, x: number, y: number, z: number) => {
   946→    const m = new THREE.Mesh(geo, mat);
   947→    m.position.set(x, y, z);
   948→    m.castShadow = true;
   949→    return m;
   950→  };
   951→
   952→  // جذع
   953→  const torso = new THREE.Group();
   954→  torso.add(mkMesh(new THREE.BoxGeometry(0.4, 0.55, 0.24), shirt, 0, 0, 0));
   955→  torso.add(mkMesh(new THREE.BoxGeometry(0.42, 0.18, 0.26), shirt, 0, -0.3, 0));
   956→  // رأس
   957→  const headG = new THREE.Group();
   958→  headG.position.y = 0.42;
   959→  headG.add(mkMesh(new THREE.BoxGeometry(0.22, 0.26, 0.24), skin, 0, 0, 0));
   960→  headG.add(mkMesh(new THREE.BoxGeometry(0.24, 0.1, 0.26), hair, 0, 0.12, -0.01));
   961→  torso.add(headG);
   962→  // ذراعان
   963→  const mkArm = (side: number) => {
   964→    const a = new THREE.Group();
   965→    a.position.set(side * 0.26, 0.18, 0);
   966→    a.add(mkMesh(new THREE.BoxGeometry(0.1, 0.3, 0.11), shirt, 0, -0.15, 0));
   967→    const fore = new THREE.Group();
   968→    fore.position.y = -0.3;
   969→    fore.add(mkMesh(new THREE.BoxGeometry(0.09, 0.3, 0.1), skin, 0, -0.15, 0));
   970→    a.add(fore);
   971→    a.userData.fore = fore;
   972→    return a;
   973→  };
   974→  const armL = mkArm(-1);
   975→  const armR = mkArm(1);
   976→  torso.add(armL, armR);
   977→  // ساقان
   978→  const mkLeg = (side: number) => {
   979→    const l = new THREE.Group();
   980→    l.add(mkMesh(new THREE.BoxGeometry(0.14, 0.42, 0.15), pants, 0, -0.21, 0));
   981→    const shin = new THREE.Group();
   982→    shin.position.y = -0.42;
   983→    shin.add(mkMesh(new THREE.BoxGeometry(0.13, 0.42, 0.14), pants, 0, -0.21, 0));
   984→    l.add(shin);
   985→    l.userData.shin = shin;
   986→    return l;
   987→  };
   988→  const legL = mkLeg(-1);
   989→  const legR = mkLeg(1);
   990→
   991→  if (pose.stance === "sit") {
   992→    torso.position.y = 0.78;
   993→    g.add(torso);
   994→    legL.position.set(-0.11, 0.5, 0);
   995→    legL.rotation.x = -1.4;
   996→    (legL.userData.shin as THREE.Group).rotation.x = 1.5;
   997→    legR.position.set(0.11, 0.5, 0);
   998→    legR.rotation.x = -1.4;

  1270→export interface DoorDef {
  1271→  group: THREE.Group;

  1278→/**
  1279→ * باب حقيقي بمفصلة تُفتح بالتفاعل — يدور حول حافته.
  1280→ * الأصل pivot عند الحافة اليسرى (نظرة من الخارج).
  1281→ */
  1282→export function makeDoor(width = 1.5, style: "wood" | "metal" | "double" = "wood"): DoorDef {
  1283→  const group = new THREE.Group();
  1284→  const h = 2.35;
  1285→  const leafW = style === "double" ? width / 2 : width;
  1286→  const mat =
  1287→    style === "metal"
  1288→      ? stdMat({ color: 0x3a4045, roughness: 0.45, metalness: 0.75 })
  1289→      : stdMat({ color: 0x4c3826, roughness: 0.7 });
  1290→
  1291→  const mkLeaf = (w: number) => {
  1292→    const leaf = new THREE.Group();
  1293→    const panel = new THREE.Mesh(new THREE.BoxGeometry(w, h, 0.07), mat);
  1294→    panel.position.set(w / 2, h / 2, 0);
  1295→    panel.castShadow = true;
  1296→    leaf.add(panel);
  1297→    // تكسير داخلي
  1298→    for (let i = 0; i < 2; i++) {
  1299→      const inset = new THREE.Mesh(new THREE.BoxGeometry(w - 0.24, h * 0.36, 0.02), style === "metal" ? mat : stdMat({ color: 0x3e2c1c, roughness: 0.75 }));
  1300→      inset.position.set(w / 2, h * (0.28 + i * 0.42), 0.045);
  1301→      leaf.add(inset);
  1302→    }
  1303→    // مقبض
  1304→    const knob = new THREE.Mesh(new THREE.SphereGeometry(0.035, 8, 8), stdMat({ color: 0x8a8f94, metalness: 0.9, roughness: 0.3 }));
  1305→    knob.position.set(w - 0.14, 1.05, 0.06);
  1306→    leaf.add(knob);
  1307→    // لوحة (للأبواب المزدوجة: شعار)
  1308→    leaf.traverse((o) => {
  1309→      if (o instanceof THREE.Mesh) o.userData.noHit = false;
  1310→    });
  1311→    return leaf;
  1312→  };
  1313→
  1314→  if (style === "double") {
  1315→    const l = mkLeaf(leafW);
  1316→    const r = mkLeaf(leafW);
  1317→    r.scale.x = -1;
  1318→    r.position.x = width;
  1319→    group.add(l, r);
  1320→  } else {
  1321→    group.add(mkLeaf(leafW));
  1322→  }
  1323→  return { group, width, open: false, locked: false, kind: style };
  1324→}

  1328→/**
  1329→ * يدمج كل دلاء PropCtx التي لم تُدمج بعد — تُستدعى مرة واحدة من world.ts
  1330→ * مع موادّها، وتضيف mesh لكل دلو إلى المشهد.
  1331→ */
  1332→export function mergePropBuckets(
  1333→  scene: THREE.Scene,
  1334→  buckets: Map<string, THREE.BufferGeometry[]>,
  1335→  mats: Record<string, THREE.Material>,
  1336→  noShadowBuckets = new Set(["glassDark", "paint", "water", "lampGlow"]),
  1337→) {
  1338→  for (const [bucket, geos] of buckets) {
  1339→    if (geos.length === 0) continue;
  1340→    const mat = mats[bucket];
  1341→    if (!mat) continue;
  1342→    const merged = mergeGeometries(geos, false);
  1343→    if (!merged) continue;
  1344→    const mesh = new THREE.Mesh(merged, mat);
  1345→    mesh.castShadow = !noShadowBuckets.has(bucket);
  1346→    mesh.receiveShadow = true;
  1347→    scene.add(mesh);
  1348→    geos.forEach((g) => g.dispose());
  1349→  }
  1350→}
  1351→

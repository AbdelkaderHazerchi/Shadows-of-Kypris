    97→// ── Canvas texture helpers ──
    98→function canvasTex(
    99→  w: number,
   100→  h: number,
   101→  draw: (ctx: CanvasRenderingContext2D) => void,
   102→  repeat = 1,
   103→): THREE.Texture {
   104→  const c = document.createElement("canvas");
   105→  c.width = w;
   106→  c.height = h;
   107→  const ctx = c.getContext("2d")!;
   108→  draw(ctx);
   109→  const tex = new THREE.CanvasTexture(c);
   110→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
   111→  tex.repeat.set(repeat, repeat);
   112→  tex.anisotropy = 4;
   113→  return tex;
   114→}

   116→function noiseFill(
   117→  ctx: CanvasRenderingContext2D,
   118→  w: number,
   119→  h: number,
   120→  base: string,
   121→  amt: number,
   122→  dark: string,
   123→) {
   124→  ctx.fillStyle = base;
   125→  ctx.fillRect(0, 0, w, h);
   126→  const img = ctx.getImageData(0, 0, w, h);
   127→  const d = img.data;
   128→  for (let i = 0; i < d.length; i += 4) {
   129→    const n = (Math.random() - 0.5) * amt;
   130→    d[i] = Math.max(0, Math.min(255, d[i] + n));
   131→    d[i + 1] = Math.max(0, Math.min(255, d[i + 1] + n));
   132→    d[i + 2] = Math.max(0, Math.min(255, d[i + 2] + n));
   133→  }
   134→  ctx.putImageData(img, 0, 0);
   135→  // stains
   136→  for (let i = 0; i < 14; i++) {
   137→    const x = Math.random() * w;
   138→    const y = Math.random() * h;
   139→    const r = 6 + Math.random() * 30;
   140→    const g = ctx.createRadialGradient(x, y, 0, x, y, r);
   141→    g.addColorStop(0, dark);
   142→    g.addColorStop(1, "rgba(0,0,0,0)");
   143→    ctx.fillStyle = g;
   144→    ctx.fillRect(x - r, y - r, r * 2, r * 2);
   145→  }
   146→}

   317→function pushStatic(ctx: BuildCtx, bucket: string, geo: THREE.BufferGeometry, matrix: THREE.Matrix4) {
   318→  const g = geo.clone();
   319→  g.applyMatrix4(matrix);
   320→  if (!ctx.staticGeos.has(bucket)) ctx.staticGeos.set(bucket, []);
   321→  ctx.staticGeos.get(bucket)!.push(g);
   322→}

   324→function addBox(
   325→  ctx: BuildCtx,
   326→  bucket: string,
   327→  x: number,
   328→  y: number,
   329→  z: number,
   330→  w: number,
   331→  h: number,
   332→  d: number,
   333→  solid = true,
   334→) {
   335→  const m = new THREE.Matrix4().compose(
   336→    new THREE.Vector3(x, y + h / 2, z),
   337→    new THREE.Quaternion(),
   338→    new THREE.Vector3(w, h, d),
   339→  );
   340→  pushStatic(ctx, bucket, new THREE.BoxGeometry(1, 1, 1), m);
   341→  if (solid) {
   342→    ctx.colliders.push({
   343→      minX: x - w / 2,
   344→      maxX: x + w / 2,
   345→      minZ: z - d / 2,
   346→      maxZ: z + d / 2,
   347→    });
   348→  }
   349→}

   351→/** جدار مع فتحات للأبواب: gaps = [{at, width}] مواقع على طول الجدار */
   352→function wallWithGaps(
   353→  ctx: BuildCtx,
   354→  bucket: string,
   355→  x1: number,
   356→  z1: number,
   357→  x2: number,
   358→  z2: number,
   359→  gaps: { at: number; width: number }[] = [],
   360→) {
   361→  const horiz = Math.abs(x2 - x1) > Math.abs(z2 - z1);
   362→  const len = horiz ? Math.abs(x2 - x1) : Math.abs(z2 - z1);
   363→  const cx = (x1 + x2) / 2;
   364→  const cz = (z1 + z2) / 2;
   365→  const segs: { s: number; e: number }[] = [];
   366→  let cur = 0;
   367→  const sorted = [...gaps].sort((a, b) => a.at - b.at);
   368→  for (const g of sorted) {
   369→    const s = g.at - g.width / 2;
   370→    if (s > cur) segs.push({ s: cur, e: s });
   371→    cur = g.at + g.width / 2;
   372→  }
   373→  if (cur < len) segs.push({ s: cur, e: len });
   374→
   375→  for (const seg of segs) {
   376→    const segLen = seg.e - seg.s;
   377→    if (segLen <= 0.05) continue;
   378→    const mid = (seg.s + seg.e) / 2;
   379→    if (horiz) {
   380→      const sx = x1 + (x2 > x1 ? mid : -mid);
   381→      addBox(ctx, bucket, sx, 0, cz, segLen, WALL_H, T);
   382→    } else {
   383→      const sz = z1 + (z2 > z1 ? mid : -mid);
   384→      addBox(ctx, bucket, cx, 0, sz, T, WALL_H, segLen);
   385→    }
   386→  }
   387→  // lintel above gaps
   388→  for (const g of sorted) {
   389→    const gapLen = g.width;
   390→    const mid = g.at;
   391→    if (horiz) {
   392→      const sx = x1 + (x2 > x1 ? mid : -mid);
   393→      addBox(ctx, bucket, sx, 2.6, cz, gapLen, WALL_H - 2.6, T, false);
   394→    } else {
   395→      const sz = z1 + (z2 > z1 ? mid : -mid);
   396→      addBox(ctx, bucket, cx, 2.6, sz, T, WALL_H - 2.6, gapLen, false);
   397→    }
   398→  }
   399→}

   426→function addFire(ctx: BuildCtx, x: number, z: number, scale = 1) {
   427→  const l = new THREE.PointLight(0xd4562a, 3.4 * scale, 16 * scale, 1.7);
   428→  l.position.set(x, 1 + scale, z);
   429→  ctx.scene.add(l);
   430→  ctx.fireLights.push({ light: l, base: 3.4 * scale });
   431→  const fireMat = new THREE.MeshBasicMaterial({
   432→    color: 0xe86830,
   433→    transparent: true,
   434→    opacity: 0.85,
   435→  });
   436→  fireMat.userData = {};
   437→  const g1 = new THREE.PlaneGeometry(0.9 * scale, 1.5 * scale);
   438→  const p1 = new THREE.Mesh(g1, fireMat);
   439→  p1.position.set(x, 0.8 * scale, z);
   440→  p1.userData.noHit = true;
   441→  const p2 = p1.clone();
   442→  p2.rotation.y = Math.PI / 2;
   443→  p2.userData.noHit = true;
   444→  ctx.scene.add(p1, p2);
   445→  // burn mark
   446→  const decal = new THREE.Mesh(
   447→    new THREE.CircleGeometry(1.6 * scale, 12),
   448→    new THREE.MeshBasicMaterial({ color: 0x090909 }),
   449→  );
   450→  decal.rotation.x = -Math.PI / 2;
   451→  decal.position.set(x, 0.03, z);
   452→  ctx.scene.add(decal);
   453→}

   476→function wreckedCar(ctx: BuildCtx, x: number, z: number, rot = 0, burned = true) {
   477→  const color = burned ? 0x141414 : [0x4a3f33, 0x37413c, 0x45343a][Math.floor(Math.random() * 3)];
   478→  const bodyMat = new THREE.MeshStandardMaterial({ color, roughness: 0.7, metalness: 0.4 });
   479→  const grp = new THREE.Group();
   480→  const body = new THREE.Mesh(new THREE.BoxGeometry(4.2, 1.05, 1.9), bodyMat);
   481→  body.position.y = 0.75;
   482→  const cabin = new THREE.Mesh(new THREE.BoxGeometry(2.1, 0.75, 1.7), bodyMat);
   483→  cabin.position.set(-0.2, 1.6, 0);
   484→  const wheelG = new THREE.CylinderGeometry(0.34, 0.34, 0.24, 10);
   485→  wheelG.rotateX(Math.PI / 2);
   486→  const wheelM = new THREE.MeshStandardMaterial({ color: 0x0d0d0d, roughness: 0.9 });
   487→  [[-1.4, 0.95], [1.4, 0.95], [-1.4, -0.95], [1.4, -0.95]].forEach(([wx, wz]) => {
   488→    const wm = new THREE.Mesh(wheelG, wheelM);
   489→    wm.position.set(wx, 0.34, wz);
   490→    grp.add(wm);
   491→  });
   492→  grp.add(body, cabin);
   493→  grp.position.set(x, 0, z);
   494→  grp.rotation.y = rot;
   495→  ctx.scene.add(grp);
   496→  // collider approx (rotated → use expanded AABB)
   497→  const ext = Math.abs(Math.sin(rot)) * 2.1 + 2.1;
   498→  const extD = Math.abs(Math.cos(rot)) * 1.0 + 1.0;
   499→  ctx.colliders.push({
   500→    minX: x - ext,
   501→    maxX: x + ext,
   502→    minZ: z - extD,
   503→    maxZ: z + extD,
   504→  });
   505→  if (burned && Math.random() < 0.3) addFire(ctx, x + 1.4, z, 0.7);
   506→}

   508→function debrisPile(ctx: BuildCtx, x: number, z: number, w: number, d: number) {
   509→  const rnd = (a: number, b: number) => a + Math.random() * (b - a);
   510→  for (let i = 0; i < Math.floor((w * d) / 6); i++) {
   511→    const bw = rnd(1.2, 3.6);
   512→    const bh = rnd(0.6, 2.4);
   513→    const bd = rnd(1.2, 3.4);
   514→    const m = new THREE.Matrix4().compose(
   515→      new THREE.Vector3(
   516→        x + rnd(-w / 2.2, w / 2.2),
   517→        bh / 2 + rnd(0, 0.8),
   518→        z + rnd(-d / 2.2, d / 2.2),
   519→      ),
   520→      new THREE.Quaternion().setFromEuler(
   521→        new THREE.Euler(rnd(-0.2, 0.2), rnd(0, Math.PI), rnd(-0.2, 0.2)),
   522→      ),
   523→      new THREE.Vector3(bw, bh, bd),
   524→    );
   525→    pushStatic(ctx, "rubble", new THREE.BoxGeometry(1, 1, 1), m);
   526→  }
   527→  ctx.colliders.push({ minX: x - w / 2, maxX: x + w / 2, minZ: z - d / 2, maxZ: z + d / 2 });
   528→  ctx.blockedRoads.push({ x, z, w: w + 2, d: d + 2 });
   529→}

   595→function shelfProp(ctx: BuildCtx, x: number, z: number, rot = 0, len = 3) {
   596→  const c = Math.cos(rot);
   597→  const s = Math.sin(rot);
   598→  const put = (ox: number, oy: number, oz: number, w: number, h: number, d: number) => {
   599→    const m = new THREE.Matrix4().compose(
   600→      new THREE.Vector3(x + ox * c - oz * s, oy, z + ox * s + oz * c),
   601→      new THREE.Quaternion().setFromEuler(new THREE.Euler(0, rot, 0)),
   602→      new THREE.Vector3(w, h, d),
   603→    );
   604→    pushStatic(ctx, "metal", new THREE.BoxGeometry(1, 1, 1), m);
   605→  };
   606→  put(0, 1, 0, len, 2, 0.5);
   607→  put(0, 0.35, 0, len, 0.07, 0.5);
   608→  put(0, 1.05, 0, len, 0.07, 0.5);
   609→  put(0, 1.75, 0, len, 0.07, 0.5);
   610→  ctx.colliders.push({
   611→    minX: x - (Math.abs(c) * len + Math.abs(s) * 0.5) / 2,
   612→    maxX: x + (Math.abs(c) * len + Math.abs(s) * 0.5) / 2,
   613→    minZ: z - (Math.abs(s) * len + Math.abs(c) * 0.5) / 2,
   614→    maxZ: z + (Math.abs(s) * len + Math.abs(c) * 0.5) / 2,
   615→  });
   616→}

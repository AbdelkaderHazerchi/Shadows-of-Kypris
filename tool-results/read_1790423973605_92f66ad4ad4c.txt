    63→    63→  size: number,
    64→    64→  base: string,
    65→    65→  amt: number,
    66→    66→  draw?: (ctx: CanvasRenderingContext2D, s: number) => void,
    67→    67→  repeat = 2,
    68→    68→): THREE.Texture {
    69→    69→  const c = document.createElement("canvas");
    70→    70→  c.width = c.height = size;
    71→    71→  const ctx = c.getContext("2d")!;
    72→    72→  ctx.fillStyle = base;
    73→    73→  ctx.fillRect(0, 0, size, size);
    74→    74→  const img = ctx.getImageData(0, 0, size, size);
    75→    75→  const d = img.data;
    76→    76→  for (let i = 0; i < d.length; i += 4) {
    77→    77→    const n = (Math.random() - 0.5) * amt;
    78→    78→    d[i] = Math.max(0, Math.min(255, d[i] + n));
    79→    79→    d[i + 1] = Math.max(0, Math.min(255, d[i + 1] + n));
    80→    80→    d[i + 2] = Math.max(0, Math.min(255, d[i + 2] + n));
    81→    81→  }
    82→    82→  ctx.putImageData(img, 0, 0);
    83→    83→  draw?.(ctx, size);
    84→    84→  const tex = new THREE.CanvasTexture(c);
    85→    85→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    86→    86→  tex.repeat.set(repeat, repeat);
    87→    87→  tex.anisotropy = 4;

   245→   245→class PB {
   246→   246→  constructor(private ctx: PropCtx, private ox = 0, private oy = 0, private oz = 0) {}
   247→   247→  /** إحداثيات محلية → عالمية */
   248→   248→  private toWorld(x: number, y: number, z: number, o: Mat4Opts = {}) {
   249→   249→    // دوران المجموعة حول Y فقط (كل الدعائم محاذاة أو مُدارة من العالم)
   250→   250→    const c = Math.cos(this.grpRy), s = Math.sin(this.grpRy);
   251→   251→    return {
   252→   252→      x: this.ox + x * c + z * s,
   253→   253→      y: this.oy + y,
   254→   254→      z: this.oz - x * s + z * c,
   255→   255→      ry: (o.ry ?? 0) + this.grpRy,
   256→   256→      rx: o.rx ?? 0,
   257→   257→      rz: o.rz ?? 0,
   258→   258→      sx: o.sx ?? 1, sy: o.sy ?? 1, sz: o.sz ?? 1,
   259→   259→    };
   260→   260→  }
   261→   261→  private grpRy = 0;
   262→   262→  group(x: number, y: number, z: number, ry = 0) { this.ox = x; this.oy = y; this.oz = z; this.grpRy = ry; return this; }
   263→   263→  box(bucket: string, x: number, y: number, z: number, w: number, h: number, d: number, o: Mat4Opts = {}) {
   264→   264→    const p = this.toWorld(x, y, z, o);
   265→   265→    this.ctx.push(bucket, BOX, mat4(p.x, p.y, p.z, { ...o, ry: p.ry, rx: p.rx, rz: p.rz, sx: w * p.sx, sy: h * p.sy, sz: d * p.sz }));
   266→   266→    return this;
   267→   267→  }
   268→   268→  /** صندوق بمركز القاع (y = قاعدة) */
   269→   269→  boxB(bucket: string, x: number, y: number, z: number, w: number, h: number, d: number, o: Mat4Opts = {}) {
   270→   270→    return this.box(bucket, x, y + h / 2, z, w, h, d, o);
   271→   271→  }
   272→   272→  cyl(bucket: string, x: number, y: number, z: number, rTop: number, rBot: number, h: number, o: Mat4Opts & { seg?: number } = {}) {
   273→   273→    const p = this.toWorld(x, y, z, o);
   274→   274→    const g = new THREE.CylinderGeometry(rTop, rBot, h, o.seg ?? 10, 1);
   275→   275→    if (o.rx || o.ry || o.rz) g.rotateX(o.rx ?? 0), g.rotateY(o.ry ?? 0), g.rotateZ(o.rz ?? 0);
   276→   276→    g.translate(p.x, p.y, p.z);
   277→   277→    this.ctx.push(bucket, g, mat4(0, 0, 0));
   278→   278→    return this;
   279→   279→  }
   280→   280→  sph(bucket: string, x: number, y: number, z: number, r: number, o: Mat4Opts = {}) {
   281→   281→    const p = this.toWorld(x, y, z, o);
   282→   282→    this.ctx.push(bucket, UNIT_SPHERE, mat4(p.x, p.y, p.z, { ...o, ry: p.ry, sx: r * p.sx, sy: r * p.sy, sz: r * p.sz }));
   283→   283→    return this;
   284→   284→  }
   285→   285→  /** استنساخ هندسة جاهزة بمصفوفة (للأنابيب المعقدة) */
   286→   286→  geo(bucket: string, geo: THREE.BufferGeometry, x: number, y: number, z: number, o: Mat4Opts = {}) {
   287→   287→    const p = this.toWorld(x, y, z, o);
   288→   288→    const g = geo.clone();
   289→   289→    if (o.rx || o.ry || o.rz) g.rotateX(o.rx ?? 0), g.rotateY(o.ry ?? 0), g.rotateZ(o.rz ?? 0);
   290→   290→    g.translate(p.x, p.y, p.z);
   291→   291→    this.ctx.push(bucket, g, mat4(0, 0, 0));
   292→   292→    return this;
   293→   293→  }
   294→   294→  /** مصادم AABB بمقاسات محلية (غير مُدار) */
   295→   295→  hit(x: number, z: number, w: number, d: number) {
   296→   296→    const c = Math.cos(this.grpRy), s = Math.sin(this.grpRy);
   297→   297→    const wx = this.ox + x * c + z * s;
   298→   298→    const wz = this.oz - x * s + z * c;
   299→   299→    const ew = Math.abs(c) * w / 2 + Math.abs(s) * d / 2;
   300→   300→    const ed = Math.abs(s) * w / 2 + Math.abs(c) * d / 2;
   301→   301→    this.ctx.collider(wx - ew, wx + ew, wz - ed, wz + ed);
   302→   302→  }
   303→   303→}

   531→   531→export function clockTower(w: PropCtx, x: number, z: number): { hands: THREE.Object3D[] } {

   533→   533→  const H = 21;
